# ==================================== FULL SETTINGS CATALOG =====================================
# Every player-facing setting, with its built-in default as the value.
#
# THIS FILE IS NOT READ BY THE MOD. It is the reference, refreshed by every update.
# To change a setting: COPY its line into halo_vr_user.cfg (next to this file), remove
# the leading #, and set your value there. halo_vr_user.cfg is read live (~2 s), updates
# never touch it, and deleting it resets everything to the defaults listed here.
# Easier still: the in-game menu (UEVR overlay > Script UI) edits the same settings with
# sliders and dropdowns -- key and button values are stored as codes in the file, but the
# menu shows them by name.
#
# Layout note (the in-game menu renders this file): the comment block right under each section
# banner, up to the first empty '#' line, is that section's summary; the comment block above
# each key is that key's own clarification.

# ===================================== CONTROL PREFERENCES ======================================
# How you turn and what the buttons do. Button names are standard gamepad names (like an Xbox
# pad). Keyboard shortcuts only work while the game window is focused.
#
# ---- Turning (aiming never turns you; the right stick does).
# 1 = snap turning (discrete steps -- most comfortable), 2 = smooth turning, 0 = off.
#turnmode=1
# Degrees per snap.
#snapdeg=45
# Smooth-turn speed, degrees per second.
#smoothdps=90
# How far the stick must move before a turn registers.
#turndz=0.30

# Hold right stick UP: the left stick becomes the d-pad (grenades / weapon switching).
#mapdpadshift=1

#mapdpaddz=0.50

#maprstickdz=0.65
# Pushing the right stick DOWN presses this button. Default: B, which is crouch.
#maprstickdown=0x2000
# Rebind one button to another: pressing `mapfrom` sends `mapto` instead. Default: the crouch

#mapfrom=0x2000

# ---- Action binds. Put one of the mod's own actions on a button of your choice.
#
# EASIEST WAY: the in-game "Halo VR Controls" panel. Press Rebind, close the menu, press the
# button -- it records what your controller actually sent. Worth doing rather than typing a
# value here, because button names differ between runtimes: on Quest the right controller's B
# arrives as "X". A wrong mask silently unbinds an action and looks like a broken mod.
#
# 0 = keep the built-in mechanism (noted per line). These ship unbound on purpose: the stick and
# gesture controls are the designed experience, and a bind is an addition, never a prerequisite.
# Crouch. 0 = right stick DOWN.
#bindcrouch=0x0000
# Melee. 0 = the swing gesture only.
#bindmelee=0x0000
# Reload. 0 = the magazine gesture only.
#bindreload=0x0000
# Scope toggle. 0 = left trigger. Binding a button KEEPS the trigger working as well.
#bindscope=0x0000
# Hold to turn the left stick into the d-pad. 0 = right stick UP. A bound button REPLACES the
# stick gesture (rather than adding to it), so binding this is also how you turn that gesture off.
#binddpadshift=0x0000
# How long an armed rebind waits for a press before giving up, milliseconds. It expires on
# purpose: a capture EATS the press it records, so an arm that never timed out would sit there
# and silently swallow whatever you pressed next.
#bindcapturems=20000

# ---- Menus. In menus, this button acts as Back (the remaps above pause automatically while
# a menu is open).
#mapmenuback=0x4000

# ---- Keyboard shortcuts. (The in-game Calibration panel does the same without a keyboard.)

# Hold to calibrate where shots go; release to save. Default: Page Down.

# ==================================== COMFORT -- VIEW & HEAD ====================================
# Your view stays put while you aim (only the gun moves), and head turning is always 1:1 --
# that is what keeps this comfortable.
#
# THE LEASH keeps you and the game's camera together: the camera cannot follow your body, so
# moving past the radius makes the world gently follow you, which keeps your aim honest.
# Radii are cm of free 1:1 movement, sideways / vertically; 0 = follow immediately (default).
# Try 30 for a small free-movement bubble. hmdleash=0 removes the leash entirely: full free
# movement, but the further you get from where the game thinks you are, the more the game's
# own flat crosshair lies -- keep the mod's reticule on.
#hmdleash=1
#hmdleashlat=0
#hmdleashvert=0

# ==================================== RETICULE (world-space) ====================================
# The mod draws the game's own crosshair out in the world, exactly where your shots will land
# -- hit marker included.
#
# 1 = show the mod's reticule.
#aimreticule=1

# Place the reticule ON the surface you are aiming at (default) -- that keeps it accurate no
# matter where your head is. 0 = float at a fixed distance instead.
#aimreticuletrace=1
# Never draw it further away than this (cm) -- past that it holds this distance, still on
# target.
#aimreticulemaxdist=600
# Lift it this far (cm) off walls so it does not sink into them.
#aimreticulesurfaceoff=50

# ---- Size. The reticule scales with distance so it always looks the same size, and it fades
# out closer than minscaledist so it cannot fill your view at point blank.
#aimreticuleminscale=0
#aimreticuleminscaledist=70
#aimreticulemaxscale=1.0

# ---- Distances, cm. Where it floats when nothing is hit:
#aimreticuledist=500
# In vehicles: pushed further out to clear the vehicle. 0 = same as on foot.
#aimreticuledistveh=2200
# Reticule size in vehicles, as a fraction of the on-foot size.
#aimreticulescaleveh=0.5

# ---- Steadiness (visual only -- never changes where shots land). Milliseconds of smoothing:
# higher = calmer but laggier, 0 = raw. Raise the second one a little if the reticule shimmers
# while you hold still.
#aimreticulesmoothms=0
#aimreticulesmoothctrlms=8

# ================================ VR WEAPON HANDLING (gestures, arms, hands) ====================
# Contributed by blindcowboy24 (PR #7). Everything here is OFF or neutral by default -- copy a
# line into halo_vr_user.cfg and uncomment it to turn a feature on.

# ---------------------------------------------------------------- MELEE BY SWING
# Swing the aim hand and the game melees. Detection runs on the game thread and
# publishes a deadline; the XInput hook only compares a clock against it, so the
# controller poll path stays free of any real work.
#
# WHY SPEED AND NOT ROTATION. The false positive to beat is a fast aim turn: the
# hand moves quickly but you meant to look, not hit. A turn is mostly rotation
# about the wrist with little travel; a strike is mostly travel. So the trigger
# is metres per second of controller movement, not degrees per second.
#
# Never fires in stick mode (vehicles, cutscenes, death) or during calibration.
#meleeswing=1

# Controller speed that counts as a strike, m/s. Conservative on purpose: a
# missed melee costs one swing, a spurious one shoves at the wrong moment.
#meleespeed=1.50

# How much of the motion must lie along your view direction, 0..1, measured as
# |dot(velocity, hmd_forward)| so a backhand counts too. 0 disables the check.
# This is what rejects the sideways sweep of a turn.
#meleefwd=0.0

# Rate the arm must be EXTENDING at, m/s -- d/dt of head-to-hand distance.
# THIS is the discriminator. A thrust grows it fast; an aim turn orbits the body
# at constant radius so it reads ~0; walking moves head and hand together, ~0.
#meleeext=1.50

# How far the hand must be from the head when it fires, metres. Stops a fast
# twitch near the chest counting as a punch -- the arm has to actually be out.
# Relaxed hand ~0.30-0.40, committed thrust past 0.60.
#meleereach=0.30

# Sanity ceiling on relative speed, m/s. The first live session logged a 26 m/s
# "swing", which no arm produces -- a tracking discontinuity. Above this the
# velocity history is dropped rather than fired on, because the next sample
# after a teleport is derived from the same bad position.
#meleemaxspeed=12.0
# Sanity ceiling on reach, metres. After a 97 m/s spike, head-to-hand distance
# read 2.45-2.51 m for several samples -- the velocity guard caught the jump but
# the POSITION stayed wrong. No human arm is this long.
#meleemaxreach=1.20

# Velocity smoothing time constant, ms. Long enough to kill tracking noise,
# short enough not to blunt the leading edge of a real strike.
#meleetau=20

# Refractory period after a melee fires, ms. Halo's melee animation is not
# interruptible, so one swing crossing the threshold on several consecutive
# ticks must still produce exactly one press.
#meleecooldown=500

# How long the synthetic button is held, ms. A single poll can land between the
# game's own input samples and be missed, so the press spans several.
#meleehold=80

# Pad mask ORed in to melee. 0x0080 = RTHUMB (right stick click), this profile's
# melee. Confirm against your own mapping with mapbtnlog=1 before changing it.
#meleemask=0x0080

# Prints every swing's peak speed and whether it fired -- the tuning instrument
# for meleespeed and meleefwd. Set it, swing a dozen times, read the log, turn
# it off.

# ---------------------------------------------------------------- TWO-ARM RECON
# Dumps the first-person skeleton -- every bone with its parent -- and reports
# which bone functions this build actually exposes. Read-only: nothing is
# written to the game, no bone is hidden, no transform is touched.
#
# Fires on the RISING EDGE (0 -> 1), so leaving it set does not re-dump every
# time the config reloads. To dump again, set it back to 0, wait ~2 s, set 1.
#
# Set it while standing in NORMAL GAMEPLAY WITH A WEAPON DRAWN. The rig does not
# exist in menus, in vehicles, during cutscenes, or for about 5 seconds after a
# level load, and the dump will just report "no rig component".
#
# This is the instrument that decides whether independent left/right arms are
# buildable. Results go to the UEVR log as BONEDUMP lines.

# ---------------------------------------------------------------- LEFT ARM HIDE
# Hides the left arm chain, so an independently tracked left hand can replace
# it. Fully reversible -- set back to 0 and the arm comes straight back.
#
# The bone dump established this is possible: HideBoneByName and
# UnHideBoneByName are both PRESENT, and Shoulder_L is the single root of all
# 34 left bones (hiding a bone hides its children). Weapon_M hangs off Chest_M
# rather than either wrist, so the gun is NOT dragged along with the arm.
#
# Right now this exists as PROOF, not as a feature: present in the reflection
# table is not the same as working on this mesh. Rig.cpp documents a relative
# rotation write that is present and silently does nothing, which is the whole
# reason rigmode 3 exists. Verify before anything gets built on top.
#armhide=0

# Which bone to hide:
#   Shoulder_L  the entire arm
#   Elbow_L     forearm and hand, upper arm stays
#   Wrist_L     just the hand
#armhidebone=Shoulder_L

# Apply to every first-person skeletal mesh, not just the arms. ON by default:
# the dump found THREE carrying this pose -- BPC_FP_SkeletalMesh_C,
# BPC_FP_TranslucentSkeletalMesh_C (the shield shell, which runs its own
# instance of the same anim blueprint) and BPC_FP_ShadowSkeletalMesh_C. Hiding
# only the arms leaves a floating shield limb and an arm-shaped shadow.
# Set 0 only to isolate which mesh is which while testing.
#armhideall=1

# ---------------------------------------------------------------- VR RELOAD
# Two-stage reload. Press reload to DROP the mag -- the game is NOT told to
# reload yet -- then physically fetch a fresh one from your belt and bring it
# to the gun. The game's reload fires at the moment the mag goes in.
#
# Halo's reload is one atomic animation with no magazine object to manipulate,
# so the physicality cannot come from the game. It comes from step one taking
# the gun away: while the mag is out the trigger is dead and you have to do
# something about it.
#
#   IDLE     -- press reload      --> MAG_OUT   (fire suppressed)
#   MAG_OUT  -- left hand at belt + grip --> MAG_HELD
#   MAG_OUT  -- press reload again --> IDLE     (cancel, mag re-seats)
#   MAG_HELD -- release grip      --> MAG_OUT   (dropped it)
#   MAG_HELD -- hand to the gun   --> reload fires, back to IDLE
#
# No timeout: it holds indefinitely. But it always releases on the kill switch,
# in vehicles/cutscenes/death, and during calibration.
#reloadvr=0

# UNCONFIRMED MASKS -- verify both with mapbtnlog=1 before trusting them. This
# profile's mapping is not obvious (mapfrom=0x2000 -> mapto=0x0100,
# mapmenuback=0x4000), and a wrong mask means reload never starts or steals a
# button you needed.
# 0x4000 = X, this game's reload.
#reloadmask=0x4000
# Left grip -- what you hold to keep hold of the magazine.
#reloadgrip=0x0100

# BELT ZONE, measured from your head (the only body reference VR gives us).
# How far BELOW head height the hand must be, metres.
#reloadbeltdrop=0.55
# ...and how close to your body's vertical axis, metres. A hand dropped at your
# side counts; one flung out sideways does not.
#reloadbeltrad=0.50

# How close the left hand must come to the aim hand to seat the mag, metres.
# Hand-to-hand, not hand-to-weapon: the gun is a separate actor whose grip
# point moves per weapon, while both controllers are always known.
#reloadjoin=0.30

# Swallow the trigger while the mag is out. This is what gives it stakes --
# 0 makes the whole gesture cosmetic.
#reloadnofire=1

# Pressing reload again while the mag is out re-seats it and aborts. OFF,
# because that is not how VR reloading works -- Onward, Pavlov, H3VR: once the
# mag is out you deal with it. A cancel is a menu concept, not a physical one,
# and a double-tap silently skipped the whole gesture.
#
# The involuntary cases are still covered without it: kill switch, vehicles,
# cutscenes, death and calibration all reset and release the trigger. A gesture
# you cannot physically complete is covered by reloadtimeout below.
#reloadcancel=0

# Seconds before an unfinished reload gives up, returns to idle and gives the
# fire trigger back. 0 disables the watchdog.
#
# This is the safety net for a gesture that cannot be completed -- an awkward
# seating position, tracking loss on your off hand, or simply changing your mind
# mid-firefight. Without it the trigger stays swallowed until you enter a
# vehicle or restart, which reads as "I can't shoot any more".
#
# It does NOT reload for you; it just gives the trigger back and says so in the
# log. 6 s is several times a completed gesture, so it will not cut a slow
# reload short.
#reloadtimeout=6.0

# Log every state transition -- the tuning instrument for the belt zone and
# join distance.

# HOW to hide. HideBoneByName reported success on all three meshes and changed
# nothing visible, so per-bone hiding may simply be inert on this build -- the
# same silent no-op Rig.cpp records for the relative rotation write.
#
#   0 = HideBoneByName        per-bone. Keeps the right arm. What we want if it works.
#   1 = SetVisibility(false)  whole component. Takes BOTH arms.
#   2 = SetHiddenInGame(true) whole component, different path to the same thing.
#
# Modes 1 and 2 are blunt, but they are the route to genuinely independent
# hands: SetBoneTransformByName is ABSENT on this build, so the game's arms
# cannot be posed at all. The only way to two free arms is to remove the
# game's pair and attach our own to the controllers. Both pass
# bPropagateToChildren=false, so the weapon actor stays where it is.
#armhidemode=2

# ---------------------------------------------------------------- OUR OWN HANDS
# With the game's first-person meshes hidden there is nothing on screen but a
# floating gun. These are the replacement: spawned meshes attached to the motion
# controllers through UObjectHook, the same mechanism the weapon rig uses.
#
# Spawned rather than borrowed because the bone dump closed every other route --
# SetBoneTransformByName is ABSENT on this build, so the game's own hand bones
# cannot be posed at all.
#
# The default mesh is an engine primitive. It is a placeholder and looks like
# one: Halo ships no standalone hand asset we can borrow. Point handmesh at
# something better if you find one.
#handsvr=0

# Object path to a StaticMesh, e.g.
#   handmesh=StaticMesh /Engine/BasicShapes/Sphere.Sphere
# Empty falls back to an engine sphere.
#handmesh=

# Uniform scale. The engine sphere is 100 cm radius, hence the small number.
#handscale=0.06

# Offset from the controller, centimetres, in the controller's own frame. The
# tracked point sits behind and below a real palm, so a mesh at the raw pose
# floats off the wrist.
#handoffx=0.0
#handoffy=0.0
#handoffz=0.0

# Also show a mesh on the AIM hand. Off by default -- the weapon already tracks
# that controller, so a second object there mostly intersects the gun.
#handshowaim=0

# ---------------------------------------------------------------- THE MAGAZINE
# Visible only while the reload state machine has one in your hand. Hidden
# rather than destroyed between reloads: spawning a component per reload would
# churn objects on the game thread for nothing.
#magshow=1
#magmesh=
#magscale=0.04
#magoffx=0.0
#magoffy=0.0
#magoffz=0.0

# How long to hold the reload button before it stops being a reload and becomes
# the game's own action, milliseconds.
#
# Both buttons this feature borrows already have jobs. The reload button is ALSO
# Interact and Enter Vehicle -- swallowing it unconditionally cost a whole
# chapter of vehicles -- and the left grip is Throw Grenade, so every magazine
# grab was lobbing a frag.
#
#   TAP  the reload button -> VR reload starts, the game never sees it
#   HOLD the reload button -> passes through, Interact / Enter Vehicle as normal
#
# The grip is only swallowed while a magazine is expected or in hand, so
# grenades work normally whenever you are not mid-reload.
#reloadhold=500

# The left grip belongs to US, not the game. It is the VR interaction button --
# magazine grabs now, weapon holding later -- and a button that throws a grenade
# when you reach for something cannot carry physical interactions.
#
# With this on THE GAME NEVER SEES THE GRIP, so grenades need rehoming below.
# Set 0 for the narrower rule: grip swallowed only mid-reload.
#gripexclusive=1

# GRENADE, moved off the grip. Pure remap: press grenadefrom, the plugin injects
# grenadeaction in its place.
#
# BOTH MASKS MUST BE MEASURED, not guessed. Controller buttons and XInput masks
# do not line up here -- your right controller B reports as 0x4000, which the
# game labels X. Set mapbtnlog=1, press your LEFT controller X, and read the
# mask; that is grenadefrom. grenadeaction should be whatever the grip used to
# report, since that is the mask the game already treats as throw.
#
# 0 disables the remap.
#grenadefrom=0x0000
#grenadeaction=0x0100

# Force hidden skeletal meshes to keep evaluating their pose.
#
# UE defaults to OnlyTickPoseWhenRendered, so a HIDDEN mesh can stop animating
# entirely -- and the weapon rig positions the gun from this mesh's
# PrimaryWeapon SOCKET. A frozen pose therefore means a frozen socket and a gun
# parked wherever the animation stopped, showing up intermittently depending on
# when the hide lands relative to the rig resolving.
#
# Set 0 to test whether the arm hide is what is moving your weapon.
#armkeeppose=1

# ---------------------------------------------------------------- PER-WEAPON OFFSETS
# One calibration cannot fit a pistol, an assault rifle and a rocket launcher --
# they do not share a grip geometry. These adjust the calibrated base PER
# WEAPON, so pose-match calibration still does the work and you only add an
# entry where a specific weapon disagrees.
#
# A weapon with no entry behaves exactly as it does now, so turning this on
# changes nothing until you tune something.
#wpnoffsets=1

# Logs which weapon you are holding and which entry matched. Turn it on once to
# harvest the class names, then off.

# wpnoff=<match>,<dx>,<dy>,<dz>,<dgrip>,<dyaw>,<droll>
#
#   match   substring of the weapon actor class -- "AssaultRifle" matches
#           BP_AssaultRifle_WeaponActor_C. First match wins, so put specific
#           entries above general ones.
#   dx/dy/dz        centimetres, ADDED to offx/offy/offz
#   dgrip/dyaw/droll  degrees, ADDED to grip/gripyaw/griproll
#
# Trailing values are optional: "wpnoff=Pistol,1.5" only nudges X.
# Repeatable, up to 24 entries. Examples, commented out:
#
# wpnoff=Pistol,0.0,0.0,-2.0
# wpnoff=AssaultRifle,0.0,0.0,1.5,-3.0
# wpnoff=RocketLauncher,0.0,-1.0,3.0

# PER-WEAPON CAPTURE KEY. Same gesture as the global calibration on END: hold
# it, the weapon freezes, line your controller up with where it should sit,
# release. The only difference is the destination -- this stores a DELTA for
# the weapon in your hands and leaves the global calibration alone.
#
# Captures are written to halo_vr_weapons.cfg, which the plugin owns and
# rewrites in full. Hand-written wpnoff lines stay here in halo_vr.cfg and keep
# their comments; both files feed the same table.
#
#   HOME=0x24  INSERT=0x2D  DELETE=0x2E  PGUP=0x21  PGDN=0x22   (0 disables)
#wpncalibkey=0x2D
